"use client"

import React, { useState, useEffect, useCallback } from "react"
import { motion, AnimatePresence } from "framer-motion"
import { Search, Swords, User } from "lucide-react"
import { Button } from "@/components/ui/button"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Progress } from "@/components/ui/progress"
import Image from "next/image"
import { useMultiplayer } from "../../hooks/useMultiplayer"
import { IUserInfo } from "../interfaces/user.interface"

// Modal components (you can replace these with your UI library components if available)
const ModalBackground = ({ children }: { children: React.ReactNode }) => (
  <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
    {children}
  </div>
)

const ModalContent = ({ children }: { children: React.ReactNode }) => (
  <div className="bg-white dark:bg-gray-800 rounded-lg p-6 relative">
    {children}
  </div>
)

const ModalTitle = ({ children }: { children: React.ReactNode }) => (
  <h2 className="text-xl font-bold mb-4">{children}</h2>
)

const CloseButton = ({ onClose }: { onClose: () => void }) => (
  <button
    onClick={onClose}
    className="absolute top-2 right-2 text-gray-500 hover:text-gray-700"
  >
    &times;
  </button>
)

// Modal component
const Modal: React.FC<{
  isOpen: boolean
  onClose: () => void
  children: React.ReactNode
}> = ({ isOpen, onClose, children }) => {
  if (!isOpen) return null

  return (
    <ModalBackground onClick={onClose}>
      <ModalContent onClick={(e) => e.stopPropagation()}>
        <CloseButton onClose={onClose} />
        {children}
      </ModalContent>
    </ModalBackground>
  )
}

interface PvpProps {
  userInfo: IUserInfo
}

export default function Pvp({ userInfo }: PvpProps) {
  const { searchPlayer, startFight, enablePvp } = useMultiplayer()
  const [showEnableModal, setShowEnableModal] = useState(false)
  const [opponent, setOpponent] = useState<any>(null)
  const [combatResult, setCombatResult] = useState<any>(null)
  const [currentRound, setCurrentRound] = useState(0)
  const [isAttacking, setIsAttacking] = useState(false)
  const [attackAnimation, setAttackAnimation] = useState({ attacker: "", defender: "" })
  const [isSearching, setIsSearching] = useState(false)
  const [player, setPlayer] = useState({
    ...userInfo,
    health: userInfo.maxHealth || 1000,
    maxHealth: userInfo.maxHealth || 1000,
    image: userInfo.image || "/placeholder.svg?height=100&width=100",
  })

  useEffect(() => {
    if (!userInfo.pvp || !userInfo.pvp.pvpEnabled) {
      setShowEnableModal(true)
    }
  }, [userInfo])

  const handleEnablePvp = async () => {
    await enablePvp()
    setShowEnableModal(false)
  }

  const handleSearch = async () => {
    setIsSearching(true)
    const players = await searchPlayer()
    if (players.length > 0) {
      const opponentData = players[0]
      setOpponent({
        ...opponentData,
        health: opponentData.maxHealth || 1000,
        maxHealth: opponentData.maxHealth || 1000,
        image: opponentData.image || "/placeholder.svg?height=100&width=100",
      })
      setPlayer((prev) => ({ ...prev, health: prev.maxHealth }))
      setCombatResult(null)
      setCurrentRound(0)
      setIsAttacking(false)
    }
    setIsSearching(false)
  }

  const handleAttack = useCallback(async () => {
    if (!opponent) return
    setIsAttacking(true)
    const result = await startFight(userInfo.id, opponent.id)
    setCombatResult(result)
    setCurrentRound(0)
    setIsAttacking(false)
  }, [opponent, startFight, userInfo.id])

  return (
    <div className="container mx-auto p-4 max-w-md">
      <h1 className="text-2xl font-bold mb-4 text-center">PVP Arena</h1>

      <div className="mb-4 flex justify-center">
        <Button onClick={handleSearch} disabled={isAttacking || isSearching}>
          <Search className="w-4 h-4 mr-2" />
          {isSearching ? "Searching..." : "Find Opponent"}
        </Button>
      </div>

      <div className="grid grid-cols-2 gap-4 mb-4">
        <PlayerCard
          player={player}
          title="You"
          isAttacking={attackAnimation.attacker === player.username}
          isDefending={attackAnimation.defender === player.username}
        />
        {opponent ? (
          <PlayerCard
            player={opponent}
            title="Opponent"
            isAttacking={attackAnimation.attacker === opponent.username}
            isDefending={attackAnimation.defender === opponent.username}
          />
        ) : (
          <Card>
            <CardHeader>
              <CardTitle>Opponent</CardTitle>
            </CardHeader>
            <CardContent className="flex flex-col items-center justify-center h-full">
              <User size={48} className="text-gray-400 mb-2" />
              <p className="text-gray-500">No opponent found</p>
            </CardContent>
          </Card>
        )}
      </div>

      <AnimatePresence>
        {opponent && !isAttacking && (
          <motion.div
            initial={{ opacity: 0, y: 20 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -20 }}
          >
            <Button
              onClick={handleAttack}
              className="w-full"
              disabled={isAttacking}
            >
              <Swords className="w-4 h-4 mr-2" />
              Attack
            </Button>
          </motion.div>
        )}
      </AnimatePresence>

      <AnimatePresence>
        {combatResult && (
          <motion.div
            initial={{ opacity: 0, scale: 0.8 }}
            animate={{ opacity: 1, scale: 1 }}
            exit={{ opacity: 0, scale: 0.8 }}
            className="mt-4"
          >
            <Card>
              <CardHeader>
                <CardTitle>Combat Result</CardTitle>
              </CardHeader>
              <CardContent>
                <p className="font-bold mt-2">Combat Ended!</p>
                <p>Winner: {combatResult.winner}</p>
                <p>Loser: {combatResult.loser}</p>
                <p>Rounds: {combatResult.rounds}</p>
                <p>Loot: ${combatResult.loot.toLocaleString()}</p>
              </CardContent>
            </Card>
          </motion.div>
        )}
      </AnimatePresence>

      {/* Modal for enabling PvP */}
      <Modal isOpen={showEnableModal} onClose={() => setShowEnableModal(false)}>
        <ModalTitle>Enable PvP</ModalTitle>
        <p>Do you want to enable PvP mode?</p>
        <Button onClick={handleEnablePvp} className="mt-4">
          Enable PvP
        </Button>
      </Modal>
    </div>
  )
}

function PlayerCard({
  player,
  title,
  isAttacking,
  isDefending,
}: {
  player: any
  title: string
  isAttacking: boolean
  isDefending: boolean
}) {
  if (!player) return null

  return (
    <Card
      className={`overflow-hidden ${
        isAttacking ? "animate-attack" : ""
      } ${isDefending ? "animate-defend" : ""}`}
    >
      <CardHeader>
        <CardTitle className="flex items-center gap-2">
          <div className="relative w-10 h-10 overflow-hidden rounded-full">
            <Image
              src={player.image || "/placeholder.svg?height=100&width=100"}
              alt={player.username}
              width={40}
              height={40}
              layout="fixed"
              objectFit="cover"
            />
          </div>
          {title}
        </CardTitle>
      </CardHeader>
      <CardContent>
        <p>Name: {player.username}</p>
        <p>Reputation: {player.reputation}</p>
        <p>Cash: ${player.cashAmount?.toLocaleString()}</p>
        <p>Victories: {player.pvp?.victory || 0}</p>
        <p>Defeats: {player.pvp?.defeat || 0}</p>
        <div className="mt-2">
          <div className="flex items-center mb-1">
            <span>
              HP: {player.health} / {player.maxHealth}
            </span>
          </div>
          <Progress
            value={(player.health / player.maxHealth) * 100}
            className="h-2"
          />
        </div>
      </CardContent>
    </Card>
  )
}
